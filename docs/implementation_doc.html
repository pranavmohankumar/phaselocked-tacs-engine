4. Core Technical Starter Skeleton

import numpy as np
from scipy.signal import hilbert, butter, filtfilt

def get_instantaneous_phase(signal, fs=1000, target_freq=(8, 12)):
    b, a = butter(2, [target_freq[0]/(fs/2), target_freq[1]/(fs/2)], btype='band')
    filtered = filtfilt(b, a, signal)
    analytic_signal = hilbert(filtered)
    phase = np.angle(analytic_signal)
    return phase # Phase in radians (-pi to pi)
